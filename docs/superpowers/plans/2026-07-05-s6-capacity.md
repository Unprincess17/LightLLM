```python
def bracket_capacity(cell_spec, workload, session, duration_s=20):
    """Binary search for the highest stable offered rate."""
    lo, hi = 10.0, 2000.0  # req/s range
    c_low = lo
    c_high = hi
    
    # Phase 1: geometric probe
    rate = lo
    while rate < hi:
        stable = run_and_check_stability(rate, duration_s)
        if stable:
            c_low = rate
            rate *= 2
        else:
            c_high = rate
            break
    
    # Phase 2: binary search to ±5%
    while (c_high - c_low) / c_low > 0.05:
        mid = (c_low + c_high) / 2
        stable = run_and_check_stability(mid, duration_s)
        if stable:
            c_low = mid
        else:
            c_high = mid
    
    return c_low, c_high
```
