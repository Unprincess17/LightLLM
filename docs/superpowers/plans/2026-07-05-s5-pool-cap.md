def check_equivalence(a_throughput, b_throughput, a_p99, b_p99):
    """TOST: throughput ratio in [0.95,1.05], P99 ratio in [0.90,1.10]."""
    t_pass = tost_equivalence(a_throughput, b_throughput, margin=0.05 * statistics.mean(a_throughput))
    p_pass = tost_equivalence(a_p99, b_p99, margin=0.10 * statistics.mean(a_p99))
    return t_pass and p_pass
